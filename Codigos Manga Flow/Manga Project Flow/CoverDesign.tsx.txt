import React, { forwardRef, useImperativeHandle } from 'react';
import { Project, toDataImageSrc } from '../types';

export interface CoverDesignerHandle {
  exportCover: () => Promise<void>;
}

export const CoverDesigner = forwardRef<
  CoverDesignerHandle,
  {
    project: Project;
    updateProject: (updates: Partial<Project>) => void;
  }
>(({ project }, ref) => {
  useImperativeHandle(ref, () => ({
    exportCover: async () => {
      window.print();
    },
  }));

  const wrapSrc = toDataImageSrc(project.cover.fullWrapBase64);

  return (
    <div className="w-full max-w-4xl flex flex-col items-center gap-6 animate-slide-in-right">
      <div className="text-center">
        <h1 className="text-[24px] font-bold tracking-tight">Paperback Wrap Designer</h1>
        <p className="text-[11px] text-white/40 font-mono mt-1">
          Matched Front &amp; Back · 6x9&quot; Standard Format · Auto-spine:{' '}
          {project.cover.spineWidthInches.toFixed(3)}&quot;
        </p>
      </div>

      <div className="relative flex flex-col w-full items-center justify-center p-8 bg-[#0e0e0e] border border-white/10 rounded-3xl">
        {wrapSrc ? (
          <div className="relative w-full aspect-[16/9] rounded-2xl overflow-hidden border border-white/20">
            <img src={wrapSrc} alt="Full paperback wrap" className="w-full h-full object-cover" />
            <div className="absolute inset-0 flex pointer-events-none">
              <div className="flex-1 relative bg-black/20 border-r border-dashed border-white/25 flex items-center justify-center">
                <span className="text-[10px] font-bold text-white/45 uppercase tracking-widest bg-black/45 px-2.5 py-1 rounded-full">
                  Back Cover
                </span>
                <div className="absolute left-4 bottom-4 w-24 bg-white/90 text-black rounded p-1.5 text-center">
                  <div
                    className="h-5 mb-1"
                    style={{
                      background:
                        'repeating-linear-gradient(90deg,#0e0e0e 0 2px,transparent 2px 4px,#0e0e0e 4px 5px,transparent 5px 8px)',
                    }}
                  />
                  <span className="block font-mono text-[7px] font-bold">ISBN 978-0-00000-000-0</span>
                </div>
              </div>
              <div className="w-8 bg-black/40 border-r border-dashed border-white/25 flex items-center justify-center">
                <span className="text-[8px] font-bold text-white/60 uppercase rotate-90 whitespace-nowrap font-mono">
                  Spine ({project.cover.spineWidthInches.toFixed(2)}&quot;)
                </span>
              </div>
              <div className="flex-1 bg-black/10 flex items-center justify-center">
                <span className="text-[10px] font-bold text-white/45 uppercase tracking-widest bg-black/45 px-2.5 py-1 rounded-full">
                  Front Cover
                </span>
              </div>
            </div>
          </div>
        ) : (
          <div className="w-full aspect-[16/9] flex flex-col items-center justify-center gap-4 text-white/20">
            <span className="material-symbols-outlined text-[40px]">auto_stories</span>
            <span className="text-[12px] font-bold uppercase tracking-[0.2em]">
              GENERATE UNIFIED WRAP IN SIDEBAR
            </span>
          </div>
        )}
      </div>

      <div className="w-full grid grid-cols-2 gap-3">
        <div className="bg-[#141414] border border-white/10 rounded-2xl p-4 flex items-start gap-3">
          <span className="material-symbols-outlined text-[20px] text-white/50 mt-0.5">palette</span>
          <div>
            <p className="text-[12px] font-bold text-white/85 mb-1">Visual Matching</p>
            <p className="text-[11px] text-white/45 leading-relaxed">
              Single-pass 16:9 wrap ensures lighting, atmosphere, and character styling flow
              continuously from back cover across the spine to the front.
            </p>
          </div>
        </div>
        <div className="bg-[#141414] border border-white/10 rounded-2xl p-4 flex items-start gap-3">
          <span className="material-symbols-outlined text-[20px] text-white/50 mt-0.5">straighten</span>
          <div>
            <p className="text-[12px] font-bold text-white/85 mb-1">Print Resolution</p>
            <p className="text-[11px] text-white/45 leading-relaxed">
              Formatted for Amazon KDP 6x9&quot; paperback trim with dynamic{' '}
              {project.cover.spineWidthInches.toFixed(3)}&quot; spine width and bleed allowance.
            </p>
          </div>
        </div>
      </div>
    </div>
  );
});

CoverDesigner.displayName = 'CoverDesigner';
