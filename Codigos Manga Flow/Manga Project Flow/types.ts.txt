export type PaperType = 'white' | 'cream';

export interface Character {
  id: string;
  name: string;
  appearance: string;
  clothing: string;
  role: string;
  referenceMediaId?: string;
  referenceBase64?: string;
}

export interface Page {
  id: string;
  base64: string;
  prompt: string;
  artStyle?: string;
  castIds: string[]; // List of character IDs present in this page
}

export interface Cover {
  frontBase64: string;
  backBase64: string;
  fullWrapBase64?: string;
  prompt: string;
  spineWidthInches: number;
}

export interface Project {
  name: string;
  author: string;
  pageCount: number;
  paperType: PaperType;
  characters: Character[];
  pages: Page[];
  cover: Cover;
}

export function splitDataUrl(raw: string): { mimeType: string; base64: string } {
  const trimmed = raw.trim();
  if (!trimmed) {
    return { mimeType: 'image/png', base64: '' };
  }
  const match = trimmed.match(/^data:([^;,]+)(?:;[^,]*)?;base64,([\s\S]*)$/i);
  if (match) {
    return {
      mimeType: match[1].trim().toLowerCase(),
      base64: match[2].replace(/\s+/g, ''),
    };
  }
  return {
    mimeType: 'image/png',
    base64: trimmed.replace(/\s+/g, ''),
  };
}

export function toDataImageSrc(raw?: string): string {
  if (!raw) return '';
  const trimmed = raw.trim();
  if (!trimmed) return '';
  if (/^(data:|blob:|https?:\/\/|\/)/i.test(trimmed)) {
    return trimmed;
  }
  return `data:image/png;base64,${trimmed.replace(/\s+/g, '')}`;
}
