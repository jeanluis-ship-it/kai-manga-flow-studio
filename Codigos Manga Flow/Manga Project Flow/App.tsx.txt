import React, { useState, useEffect, useRef } from 'react';
import { Flow } from 'flow-sdk';
import { 
  SectionLabel, 
  PillButton, 
  FieldDropdown, 
  TextInput, 
  SegmentedToggle
} from './components/Primitives';
import { CharacterLibrary } from './components/CharacterLibrary';
import { PageEditor } from './components/PageEditor';
import { CoverDesigner } from './components/CoverDesigner';
import { ProjectManager } from './components/ProjectManager';
import { Project, Character } from './types';
import JSZip from 'jszip';
export default function App() {
  const [view, setView] = useState<'cover' | 'pages' | 'characters' | 'project'>('project');
  const [project, setProject] = useState<Project>({
    name: 'Untitled Manga',
    author: 'Author Name',
    pageCount: 100,
    paperType: 'white',
    characters: [],
    pages: [{ id: crypto.randomUUID(), base64: '', prompt: '', castIds: [] }],
    cover: { frontBase64: '', backBase64: '', prompt: '', spineWidthInches: 0.225 }
  });
  const [activePageIndex, setActivePageIndex] = useState<number>(0);
  const [isGenerating, setIsGenerating] = useState(false);
  const [status, setStatus] = useState('');
  
  const coverDesignerRef = useRef<{ exportCover: () => Promise<void> }>(null);
  useEffect(() => {
    const id = 'flow-manga-css';
    if (document.getElementById(id)) return;
    const style = document.createElement('style');
    style.id = id;
    style.textContent = `
      html, body, #root { background: #0e0e0e; color: white; height: 100%; margin: 0; font-family: 'Google Sans Text', sans-serif; overflow: hidden; }
      .manga-grid { display: grid; grid-template-columns: 300px 1fr 300px; height: 100%; }
      .no-scrollbar::-webkit-scrollbar { display: none; }
      .dark-scrollbar::-webkit-scrollbar { width: 6px; }
      .dark-scrollbar::-webkit-scrollbar-track { background: transparent; }
      .dark-scrollbar::-webkit-scrollbar-thumb { background: #595959; border-radius: 9999px; }
      @keyframes pulse-slow { 0%, 100% { opacity: 1; } 50% { opacity: 0.5; } }
      .animate-pulse-slow { animation: pulse-slow 2s cubic-bezier(0.4, 0, 0.6, 1) infinite; }
      @keyframes slide-in-right { from { transform: translateX(20px); opacity: 0; } to { transform: translateX(0); opacity: 1; } }
      .animate-slide-in-right { animation: slide-in-right 0.3s ease-out; }
    `;
    document.head.appendChild(style);
  }, []);
  const updateProject = (updates: Partial<Project>) => {
    setProject(prev => {
      const newState = { ...prev, ...updates };
      const multi = newState.paperType === 'white' ? 0.002252 : 0.0025;
      newState.cover.spineWidthInches = Math.max(0.1, newState.pageCount * multi);
      return newState;
    });
  };
  const handleDownloadAll = async () => {
    setStatus('Preparing KDP Package...');
    const zip = new JSZip();
    zip.file('kdp_manifest.json', JSON.stringify({ ...project, printSpec: "6x9 Paperback", bleed: "0.125in" }, null, 2));
    project.pages.forEach((page, i) => {
      if (page.base64) zip.file(`interior/page_${String(i + 1).padStart(3, '0')}.png`, page.base64, { base64: true });
    });
    const content = await zip.generateAsync({ type: 'blob' });
    const reader = new FileReader();
    reader.onload = async () => {
      const base64 = (reader.result as string).split(',')[1];
      await Flow.download({ base64, mimeType: 'application/zip', filename: `${project.name.replace(/\s+/g, '_')}_KDP_BUNDLE.zip` });
      setStatus('');
    };
    reader.readAsDataURL(content);
  };
  return (
    <div className="manga-grid">
      <div className="border-r border-[rgba(218,220,224,0.15)] flex flex-col p-[10px] gap-6 overflow-y-auto no-scrollbar">
        <div className="flex flex-col gap-2">
          <SectionLabel>KDP Production</SectionLabel>
          <div className="flex flex-col gap-1.5">
            <NavItem active={view === 'project'} icon="settings_print" label="Print Setup" onClick={() => setView('project')} />
            <NavItem active={view === 'characters'} icon="face" label="Character Library" onClick={() => setView('characters')} />
            <NavItem active={view === 'cover'} icon="menu_book" label="Paperback Cover" onClick={() => setView('cover')} />
            <NavItem active={view === 'pages'} icon="auto_stories" label="Interior Pages" onClick={() => setView('pages')} />
          </div>
        </div>
        {view === 'pages' && (
          <div className="flex flex-col gap-2 animate-slide-in-right">
            <SectionLabel>Manuscript</SectionLabel>
            <div className="flex flex-col gap-1.5 max-h-[300px] overflow-y-auto dark-scrollbar pr-1">
              {project.pages.map((p, i) => (
                <button 
                  key={i} 
                  onClick={() => setActivePageIndex(i)}
                  className={`flex items-center justify-between p-2 rounded-xl text-[12px] font-medium transition-all ${activePageIndex === i ? 'bg-[#969696] text-black' : 'text-white/60 hover:bg-white/5'}`}
                >
                  <span className="flex items-center gap-2">
                    <span className="text-[10px] opacity-40 font-mono">{String(i + 1).padStart(2, '0')}</span>
                    <span>Page {i + 1}</span>
                  </span>
                  <span className="material-symbols-outlined text-[16px] opacity-40">{p.base64 ? 'verified' : 'ink_high'}</span>
                </button>
              ))}
              <PillButton variant="outline" icon={<span className="material-symbols-outlined text-[18px]">add</span>} onClick={() => {
                const newPages = [...project.pages, { id: crypto.randomUUID(), base64: '', prompt: '', castIds: [] }];
                updateProject({ pages: newPages });
                setActivePageIndex(newPages.length - 1);
              }}>
                Add Page
              </PillButton>
            </div>
          </div>
        )}
        <div className="mt-auto flex flex-col gap-2">
          <SectionLabel>Submission</SectionLabel>
          <PillButton variant="solid" icon={<span className="material-symbols-outlined text-[18px]">publish</span>} onClick={handleDownloadAll}>
            Build KDP Bundle
          </PillButton>
          {status && <div className="text-[10px] text-[#969696] text-center animate-pulse-slow">{status}</div>}
        </div>
      </div>
      <div className="flex-1 bg-[#050505] relative overflow-hidden flex flex-col">
        <div className="flex-1 overflow-y-auto p-12 flex items-start justify-center">
          {view === 'pages' && <PageEditor project={project} pageIndex={activePageIndex} updateProject={updateProject} />}
          {view === 'characters' && <CharacterLibrary project={project} updateProject={updateProject} isGenerating={isGenerating} setIsGenerating={setIsGenerating} />}
          {view === 'cover' && <CoverDesigner ref={coverDesignerRef} project={project} updateProject={updateProject} />}
          {view === 'project' && <ProjectManager project={project} updateProject={updateProject} />}
        </div>
      </div>
      <div className="border-l border-[rgba(218,220,224,0.15)] flex flex-col p-[10px] gap-6">
         <ViewControls 
            view={view} 
            project={project} 
            updateProject={updateProject} 
            pageIndex={activePageIndex}
            isGenerating={isGenerating}
            setIsGenerating={setIsGenerating}
            onExportCover={() => coverDesignerRef.current?.exportCover()}
          />
      </div>
    </div>
  );
}
function NavItem({ active, icon, label, onClick }: { active: boolean; icon: string; label: string; onClick: () => void }) {
  return (
    <button 
      onClick={onClick}
      className={`flex items-center gap-3 h-[40px] px-3 rounded-xl transition-all ${active ? 'bg-white/10 text-white border border-white/10' : 'text-white/40 hover:text-white hover:bg-white/5'}`}
    >
      <span className="material-symbols-outlined text-[20px]">{icon}</span>
      <span className="text-[13px] font-medium tracking-wide">{label}</span>
    </button>
  );
}
function ViewControls({ view, project, updateProject, pageIndex, isGenerating, setIsGenerating, onExportCover }: any) {
  const [prompt, setPrompt] = useState('');
  const [style, setStyle] = useState('Classic Shonen');
  
  const generatePage = async () => {
    setIsGenerating(true);
    try {
      const currentPage = project.pages[pageIndex];
      const cast = project.characters.filter((c: Character) => currentPage.castIds.includes(c.id));
      const charVisuals = cast.map((c: Character) => `${c.name}: ${c.appearance}`).join('. ');
      const references = cast.filter((c: Character) => c.referenceMediaId).map((c: Character) => c.referenceMediaId!);
      const finalPrompt = `Professional manga page, high resolution, ${style} style, monochromatic ink, screentones. Scene: ${prompt}. Featuring: ${charVisuals}`;
      
      const result = await Flow.generate.image({
        prompt: finalPrompt,
        modelDisplayName: '🍌 Nano Banana Pro',
        aspectRatio: '3:4',
        referenceImageMediaIds: references.length > 0 ? references : undefined
      });
      const newPages = [...project.pages];
      newPages[pageIndex] = { ...newPages[pageIndex], base64: result.base64, prompt: prompt };
      updateProject({ pages: newPages });
    } catch (e) {
      console.error(e);
    } finally {
      setIsGenerating(false);
    }
  };
  const generateUnifiedCover = async () => {
    setIsGenerating(true);
    try {
      const charVisuals = project.characters.map((c: Character) => `${c.name}: ${c.appearance}`).join('. ');
      const references = project.characters.filter((c: Character) => c.referenceMediaId).map((c: Character) => c.referenceMediaId!);
      const finalPrompt = `Professional cinematic manga wrap-around cover. Right side (front): main illustration showing ${prompt}. Left side (back): matching background elements, minimalist composition. Spine in middle. Style: ${style}. Vibrant colors. Character details: ${charVisuals}`;
      
      const result = await Flow.generate.image({
        prompt: finalPrompt,
        modelDisplayName: '🍌 Nano Banana Pro',
        aspectRatio: '16:9',
        referenceImageMediaIds: references.length > 0 ? references : undefined
      });
      // For preview purposes, we'll use the full wrap and also try to extract or just display it unified
      updateProject({ 
        cover: { 
          ...project.cover, 
          fullWrapBase64: result.base64,
          frontBase64: result.base64, // Fallback preview
          backBase64: result.base64, // Fallback preview
          prompt 
        } 
      });
    } catch (e) {
      console.error(e);
    } finally {
      setIsGenerating(false);
    }
  };
  if (view === 'pages') {
    const currentPage = project.pages[pageIndex];
    return (
      <div className="flex flex-col gap-6 animate-slide-in-right">
        <div className="flex flex-col gap-2">
          <SectionLabel>Interior Settings</SectionLabel>
          <FieldDropdown 
            label="Art Style" 
            value={style} 
            options={['Classic Shonen', 'Dark Seinen', 'Modern Shojo', 'Retro Gekiga', 'Cyberpunk Manga']} 
            onChange={setStyle} 
          />
          
          <div className="mt-2">
            <SectionLabel>Active Cast ({currentPage?.castIds?.length || 0})</SectionLabel>
            <div className="flex flex-wrap gap-1 mt-1">
              {project.characters.map(c => (
                <button
                  key={c.id}
                  onClick={() => {
                    const newPages = [...project.pages];
                    const cast = new Set(newPages[pageIndex].castIds);
                    if (cast.has(c.id)) cast.delete(c.id);
                    else cast.add(c.id);
                    newPages[pageIndex].castIds = Array.from(cast);
                    updateProject({ pages: newPages });
                  }}
                  className={`px-2 py-1 rounded-lg text-[10px] font-bold border transition-all ${currentPage?.castIds?.includes(c.id) ? 'bg-[#969696] text-black border-[#969696]' : 'border-white/10 text-white/40'}`}
                >
                  {c.name}
                </button>
              ))}
            </div>
          </div>
          <TextInput 
            value={prompt} 
            onChange={setPrompt} 
            placeholder="Describe the action and panels..." 
          />
        </div>
        <PillButton 
          variant="solid" 
          icon={<span className="material-symbols-outlined text-[18px]">{isGenerating ? 'refresh' : 'auto_awesome'}</span>}
          onClick={generatePage}
          disabled={isGenerating}
        >
          {isGenerating ? 'Drawing Page...' : 'Generate High-Res Ink'}
        </PillButton>
      </div>
    );
  }
  if (view === 'cover') {
    return (
      <div className="flex flex-col gap-6 animate-slide-in-right">
        <div className="flex flex-col gap-2">
          <SectionLabel>Unified Cover Generation</SectionLabel>
          <p className="text-[11px] text-white/40 px-2 leading-relaxed">Generates a wide 16:9 wrap that ensures the front and back styles match perfectly.</p>
          <TextInput 
            value={prompt} 
            onChange={setPrompt} 
            placeholder="Describe the overall cover theme and mood..." 
          />
        </div>
        
        <div className="flex flex-col gap-1.5">
          <PillButton 
            variant="solid" 
            icon={<span className="material-symbols-outlined text-[18px]">{isGenerating ? 'refresh' : 'auto_awesome'}</span>}
            onClick={generateUnifiedCover}
            disabled={isGenerating}
          >
            {isGenerating ? 'Generating Wrap...' : `Generate Full Wrap`}
          </PillButton>
          
          <div className="pt-4 mt-4 border-t border-white/10 flex flex-col gap-2">
            <SectionLabel>Production</SectionLabel>
            <PillButton 
              variant="outline" 
              icon={<span className="material-symbols-outlined text-[18px]">print</span>}
              onClick={onExportCover}
            >
              Export KDP PDF-Ready
            </PillButton>
          </div>
        </div>
      </div>
    );
  }
  return (
    <div className="flex flex-col items-center justify-center h-full text-white/20 text-[11px] text-center px-4 italic">
      <span className="material-symbols-outlined text-[32px] mb-2 opacity-10">info</span>
      Select a view from the sidebar to access generation controls.
    </div>
  );
}
