import React, { useState } from 'react';
import { Project } from '../types';
import { SectionLabel, PillButton } from './Primitives';

export const PageEditor: React.FC<{
  project: Project;
  pageIndex: number;
  updateProject: (updates: Partial<Project>) => void;
}> = ({ project, pageIndex, updateProject }) => {
  const [showGuides, setShowGuides] = useState(false);
  const page = project.pages[pageIndex];

  const handleDelete = () => {
    if (project.pages.length <= 1) return;
    const nextPages = project.pages.filter((_, idx) => idx !== pageIndex);
    updateProject({ pages: nextPages });
  };

  return (
    <div className="w-full max-w-[540px] flex flex-col items-center gap-6 animate-slide-in-right">
      <div className="w-full flex items-center justify-between">
        <div>
          <h2 className="text-[18px] font-bold">Interior View</h2>
          <p className="text-[10px] font-mono text-white/40 uppercase">
            PAGE {pageIndex + 1} / {project.pages.length}
          </p>
        </div>
        <button
          onClick={() => setShowGuides(!showGuides)}
          className="px-3.5 py-1.5 rounded-full border border-white/15 bg-white/5 text-[10px] font-bold uppercase tracking-wider flex items-center gap-1.5"
        >
          <span className="material-symbols-outlined text-[14px]">grid_on</span>
          <span>{showGuides ? 'HIDE GUIDES' : 'SHOW GUIDES'}</span>
        </button>
      </div>

      <div className="w-full aspect-[3/4] bg-white rounded-sm relative overflow-hidden flex items-center justify-center shadow-2xl">
        {showGuides && (
          <div className="absolute inset-4 border border-dashed border-rose-500 pointer-events-none z-20" />
        )}
        {page?.base64 ? (
          <img src={`data:image/png;base64,${page.base64}`} className="w-full h-full object-cover" />
        ) : (
          <div className="flex flex-col items-center gap-2 text-black/20 select-none">
            <span className="material-symbols-outlined text-[28px]">draw</span>
            <span className="text-[12px] font-bold uppercase tracking-[0.25em]">INK AWAITING</span>
          </div>
        )}
      </div>

      <div className="w-full bg-[#141414] border border-white/10 rounded-2xl p-4 flex flex-col gap-3">
        <SectionLabel>PAGE PROMPT</SectionLabel>
        <p className="text-[12px] text-white/60 px-2">
          {page?.prompt || 'No instructions provided. Use the sidebar to generate ink.'}
        </p>
        <div className="grid grid-cols-2 gap-2 pt-1">
          <PillButton variant="outline" onClick={handleDelete}>Delete Page</PillButton>
          <PillButton variant="outline" onClick={() => window.print()}>Download Asset</PillButton>
        </div>
      </div>
    </div>
  );
};
